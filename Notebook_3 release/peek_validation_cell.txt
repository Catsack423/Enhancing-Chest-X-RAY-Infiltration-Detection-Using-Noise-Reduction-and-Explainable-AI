# Peek at VALIDATION metrics of a saved CNN checkpoint mid-training. Read-only: writes nothing, never touches test.
# Stop the CNN cell first (the unfinished epoch restarts from last.pt), run this, then rerun the CNN cell to resume.
import torch
from shared.models import load_cnn
from shared.preprocessing import ReleasePreprocessor
from shared.training import ImageDataset, loader, predict
from shared.evaluation import classification_metrics

PEEK_SEED = 42
PEEK_CONDITION = "baseline"
PEEK_CHECKPOINT = "best.pt"  # or a snapshot such as "epoch_002.pt"

exp, manifests = runs[PEEK_SEED]
validation = manifests["validation"]
epoch = exp.load_checkpoint(exp.results("CNN", PEEK_CONDITION) / PEEK_CHECKPOINT, PEEK_CONDITION)["epoch"]
prep = ReleasePreprocessor(exp, PEEK_CONDITION, DEVICE)
prep.prepare([validation])
model = load_cnn(exp, PEEK_CONDITION, DEVICE, PEEK_CHECKPOINT)
probabilities = predict(model, loader(ImageDataset(validation, prep), 64, exp.config["seed"]), DEVICE)
metrics = classification_metrics(validation.label.to_numpy(), probabilities, exp.config["cnn"]["classification_threshold"])
print(f"seed {PEEK_SEED} | {PEEK_CONDITION} | {PEEK_CHECKPOINT} (epoch {epoch}) | validation n={metrics['n']:,}")
for name in ("roc_auc", "pr_auc", "balanced_accuracy", "accuracy", "recall", "specificity", "precision", "f1"):
    print(f"  {name:<18} {metrics[name]:.4f}")
print(f"  prevalence         {metrics['positive_prevalence']:.4f}  (PR-AUC of a constant score)")
print("  confusion [[TN, FP], [FN, TP]] =", metrics["confusion_matrix"])
del model
torch.cuda.empty_cache()
